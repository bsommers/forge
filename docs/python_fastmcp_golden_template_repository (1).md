## 3. `src/auth.py` `[LOCKED]`

Manifest-driven `httpx.Auth` factory. Enforces the MCP `2026-07-28` **Token Passthrough Prohibition**: incoming MCP client headers are never forwarded upstream; upstream credentials come strictly from server-bound environment variables.

```python
from __future__ import annotations

import json
import os
import time
from pathlib import Path
from typing import TYPE_CHECKING, Any

import httpx

if TYPE_CHECKING:
    from collections.abc import AsyncGenerator, Generator

MANIFEST_PATH = Path(__file__).resolve().parent.parent / "forge-manifest.json"


class ManifestConfigError(RuntimeError):
    """Raised when required configuration or secret environment variables are missing."""


def load_manifest(path: Path = MANIFEST_PATH) -> dict[str, Any]:
    return json.loads(path.read_text(encoding="utf-8"))  # type: ignore[no-any-return]


def _require_env(env_name: str) -> str:
    value = os.environ.get(env_name, "").strip()
    if not value:
        raise ManifestConfigError(
            f"Missing required credential environment variable: {env_name}"
        )
    return value


class HeaderTokenAuth(httpx.Auth):
    """Injects Bearer or custom header credentials without logging secret values."""

    def __init__(self, header_name: str, header_value: str) -> None:
        self._header_name = header_name
        self._header_value = header_value

    def auth_flow(self, request: httpx.Request) -> Generator[httpx.Request, httpx.Response, None]:
        request.headers[self._header_name] = self._header_value
        yield request


class QueryApiKeyAuth(httpx.Auth):
    """Injects an API key query parameter."""

    def __init__(self, param_name: str, api_key: str) -> None:
        self._param_name = param_name
        self._api_key = api_key

    def auth_flow(self, request: httpx.Request) -> Generator[httpx.Request, httpx.Response, None]:
        request.url = request.url.copy_merge_params({self._param_name: self._api_key})
        yield request


class OAuth2ClientCredentialsAuth(httpx.Auth):
    """Async OAuth2 Client Credentials flow with in-memory expiry caching."""

    def __init__(self, token_url: str, client_id: str, client_secret: str, scope: str = "") -> None:
        self._token_url = token_url
        self._client_id = client_id
        self._client_secret = client_secret
        self._scope = scope
        self._access_token: str | None = None
        self._expires_at: float = 0.0

    async def _fetch_token(self) -> str:
        payload: dict[str, str] = {"grant_type": "client_credentials"}
        if self._scope:
            payload["scope"] = self._scope
        async with httpx.AsyncClient(timeout=10.0) as token_client:
            resp = await token_client.post(
                self._token_url,
                data=payload,
                auth=(self._client_id, self._client_secret),
            )
            resp.raise_for_status()
            body = resp.json()
            self._access_token = str(body["access_token"])
            expires_in = float(body.get("expires_in", 3600))
            self._expires_at = time.monotonic() + max(expires_in - 60.0, 30.0)
            return self._access_token

    async def async_auth_flow(
        self, request: httpx.Request
    ) -> AsyncGenerator[httpx.Request, httpx.Response]:
        if not self._access_token or time.monotonic() >= self._expires_at:
            await self._fetch_token()
        request.headers["Authorization"] = f"Bearer {self._access_token}"
        response = yield request
        if response.status_code == 401:
            await self._fetch_token()
            request.headers["Authorization"] = f"Bearer {self._access_token}"
            yield request


def build_auth_from_manifest(manifest: dict[str, Any]) -> httpx.Auth | None:
    auth_cfg: dict[str, Any] = manifest.get("auth", {"type": "none"})
    auth_type = str(auth_cfg.get("type", "none")).lower()

    if auth_type == "none":
        return None
    if auth_type == "bearer":
        token = _require_env(str(auth_cfg["token_env"]))
        return HeaderTokenAuth("Authorization", f"Bearer {token}")
    if auth_type == "api_key_header":
        key = _require_env(str(auth_cfg["key_env"]))
        header_name = str(auth_cfg.get("header_name", "X-API-Key"))
        return HeaderTokenAuth(header_name, key)
    if auth_type == "api_key_query":
        key = _require_env(str(auth_cfg["key_env"]))
        param_name = str(auth_cfg.get("param_name", "api_key"))
        return QueryApiKeyAuth(param_name, key)
    if auth_type == "basic":
        username = _require_env(str(auth_cfg["username_env"]))
        password = _require_env(str(auth_cfg["password_env"]))
        return httpx.BasicAuth(username=username, password=password)
    if auth_type == "oauth2_client_credentials":
        token_url = _require_env(str(auth_cfg["token_url_env"]))
        client_id = _require_env(str(auth_cfg["client_id_env"]))
        client_secret = _require_env(str(auth_cfg["client_secret_env"]))
        scope = str(auth_cfg.get("scope", ""))
        return OAuth2ClientCredentialsAuth(token_url, client_id, client_secret, scope)

    raise ManifestConfigError(f"Unsupported auth.type in forge-manifest.json: {auth_type}")
```

## 4. `src/deps.py` `[LOCKED]`

Provides FastMCP v4.2 `Depends()` injection for the shared upstream `httpx.AsyncClient` and maps upstream failures into MCP `2026-07-28` tool execution errors (`isError: true`, `resultType: "complete"`) with actionable guidance for the calling model.

```python
from __future__ import annotations

from typing import Annotated, Any

import httpx
from fastmcp.dependencies import CurrentContext, Depends
from fastmcp.exceptions import ToolError
from fastmcp.server.context import Context


def get_http_client(ctx: Context = CurrentContext()) -> httpx.AsyncClient:
    """Resolve the process-lifespan HTTP client without exposing it in JSON Schema."""
    lifespan_ctx: dict[str, Any] = ctx.lifespan_context
    client = lifespan_ctx.get("http_client")
    if not isinstance(client, httpx.AsyncClient):
        raise ToolError("Server HTTP client is unavailable; lifespan was not initialized.")
    return client


# Type alias used by agent-generated tools in src/tools/*.py
ApiClient = Annotated[httpx.AsyncClient, Depends(get_http_client)]


async def request_json(
    client: httpx.AsyncClient,
    method: str,
    path: str,
    *,
    params: dict[str, Any] | None = None,
    json_body: dict[str, Any] | list[Any] | None = None,
) -> Any:
    """Execute an upstream HTTP call and map failures to actionable MCP ToolErrors (isError=true)."""
    clean_params = {k: v for k, v in (params or {}).items() if v is not None}
    try:
        response = await client.request(
            method=method.upper(),
            url=path,
            params=clean_params or None,
            json=json_body,
        )
        response.raise_for_status()
    except httpx.TimeoutException as exc:
        raise ToolError(
            f"Upstream API timed out on {method.upper()} {path}. Try narrowing query filters or retrying."
        ) from exc
    except httpx.HTTPStatusError as exc:
        status = exc.response.status_code
        if status in (401, 403):
            raise ToolError(
                f"Upstream authorization failed (HTTP {status}) for {path}. Check server credentials/scopes."
            ) from exc
        if status == 404:
            raise ToolError(
                f"Resource not found (HTTP 404) at {path}. Verify the identifier exists."
            ) from exc
        if status == 429:
            retry_after = exc.response.headers.get("Retry-After", "unknown")
            raise ToolError(
                f"Upstream rate limit exceeded (HTTP 429). Wait before retrying (Retry-After: {retry_after})."
            ) from exc
        snippet = exc.response.text[:300].strip()
        raise ToolError(
            f"Upstream API error (HTTP {status}) on {method.upper()} {path}: {snippet}"
        ) from exc
    except httpx.RequestError as exc:
        raise ToolError(
            f"Upstream network error on {method.upper()} {path}: {type(exc).__name__}"
        ) from exc

    if response.status_code == 204 or not response.content:
        return {"status": "success", "http_status": response.status_code}
    return response.json()
```

## 5. `src/server.py` `[LOCKED]`

Configures strict `stderr` JSON logging (aligned with `SEP-2577` deprecation of protocol-level `logging/setLevel`), process-level `@lifespan` HTTP connection pooling (stateless across MCP requests per `SEP-2567`/`SEP-2575`), `ResponseLimitingMiddleware`, and deterministic `FileSystemProvider` discovery.

```python
import logging
import os
import sys
from datetime import UTC, datetime
from pathlib import Path
from typing import TYPE_CHECKING, Any

import httpx
from fastmcp import FastMCP
from fastmcp.server.lifespan import lifespan
from fastmcp.server.middleware.response_limiting import ResponseLimitingMiddleware
from fastmcp.server.providers import FileSystemProvider

from src.auth import build_auth_from_manifest, load_manifest

if TYPE_CHECKING:
    from collections.abc import AsyncIterator


class StderrJsonFormatter(logging.Formatter):
    """Structured JSON logs emitted strictly to stderr per MCP 2026-07-28 stdio & SEP-2577 guidance."""

    def format(self, record: logging.LogRecord) -> str:
        payload = {
            "ts": datetime.now(UTC).isoformat(),
            "level": record.levelname,
            "logger": record.name,
            "msg": record.getMessage(),
        }
        if record.exc_info:
            payload["exc"] = self.formatException(record.exc_info)
        return json.dumps(payload)


def configure_stderr_logging() -> None:
    handler = logging.StreamHandler(stream=sys.stderr)
    handler.setFormatter(StderrJsonFormatter())
    root = logging.getLogger()
    root.handlers.clear()
    root.addHandler(handler)
    root.setLevel(os.environ.get("LOG_LEVEL", "INFO").upper())


configure_stderr_logging()
logger = logging.getLogger("forged_mcp")

MANIFEST: dict[str, Any] = load_manifest()
TOOLS_DIR = Path(__file__).resolve().parent / "tools"
TOOLS_PROVIDER = FileSystemProvider(root=TOOLS_DIR, reload=False)


@lifespan
async def server_lifespan(_server: FastMCP) -> AsyncIterator[dict[str, Any]]:
    if TOOLS_PROVIDER.failed_files:
        raise RuntimeError(
            f"Refusing startup due to broken tool modules: {TOOLS_PROVIDER.failed_files}"
        )

    upstream = MANIFEST.get("upstream", {})
    base_url_env = str(upstream.get("base_url_env", ""))
    base_url = (
        os.environ.get(base_url_env, "").strip() if base_url_env else ""
    ) or str(upstream.get("default_base_url", "https://localhost"))
    timeout_sec = float(upstream.get("timeout_seconds", 15.0))
    auth_handler = build_auth_from_manifest(MANIFEST)

    limits = httpx.Limits(max_keepalive_connections=20, max_connections=50)
    transport = httpx.AsyncHTTPTransport(retries=2)

    async with httpx.AsyncClient(
        base_url=base_url,
        auth=auth_handler,
        timeout=httpx.Timeout(timeout_sec),
        limits=limits,
        transport=transport,
        headers={
            "Accept": "application/json",
            "User-Agent": f"forged-mcp/{MANIFEST['server_name']}-{MANIFEST['version']}",
        },
    ) as http_client:
        logger.info("Initialized stateless upstream HTTP pool for %s", MANIFEST["server_name"])
        yield {"http_client": http_client, "manifest": MANIFEST}


def create_server() -> FastMCP:
    server = FastMCP(
        name=str(MANIFEST["server_name"]),
        version=str(MANIFEST.get("version", "0.1.0")),
        instructions=str(MANIFEST.get("description", "")),
        providers=[TOOLS_PROVIDER],
        lifespan=server_lifespan,
    )
    max_bytes = int(MANIFEST.get("upstream", {}).get("max_response_bytes", 32_768))
    server.add_middleware(ResponseLimitingMiddleware(max_size=max_bytes))
    return server


mcp = create_server()


def main() -> None:
    transport_mode = os.environ.get("MCP_TRANSPORT", "stdio").lower()
    if transport_mode == "http":
        port = int(os.environ.get("PORT", "8000"))
        mcp.run(transport="http", host="0.0.0.0", port=port)  # noqa: S104
    else:
        mcp.run(transport="stdio")


if __name__ == "__main__":
    main()
```
